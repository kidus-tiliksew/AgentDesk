// MARK: Scratch diagnostics, not in the retained harness

@MainActor var diagOnFail: ((String) -> Void)?
@MainActor var diagExtended: Set<String> = []

@MainActor
func diagEmit(_ text: String) {
  print("DIAG \\(text)")
  guard let path = ProcessInfo.processInfo.environment["DIAG_CAPTURE_FILE"],
    let data = (text + "\\n").data(using: .utf8)
  else { return }
  if !FileManager.default.fileExists(atPath: path) { FileManager.default.createFile(atPath: path, contents: nil) }
  if let handle = FileHandle(forWritingAtPath: path) {
    handle.seekToEndOfFile()
    handle.write(data)
    try? handle.close()
  }
}

@MainActor
func diagSession() -> String {
  let session = (CGSessionCopyCurrentDictionary() as? [String: Any]) ?? [:]
  let locked = session["CGSSessionScreenIsLocked"].map { "\\($0)" } ?? "absent"
  func idle(_ type: CGEventType) -> String {
    String(format: "%.1f", CGEventSource.secondsSinceLastEventType(.combinedSessionState, eventType: type))
  }
  return "locked=\\(locked) displayAsleep=\\(CGDisplayIsAsleep(CGMainDisplayID())) idleAny=\\(idle(CGEventType(rawValue: ~0)!)) idleClick=\\(idle(.leftMouseDown)) idleKey=\\(idle(.keyDown)) idleMove=\\(idle(.mouseMoved)) idleScroll=\\(idle(.scrollWheel)) appActive=\\(NSApp.isActive) activations=\\(ActivationWatch.shared.count) frontmost=\\(NSWorkspace.shared.frontmostApplication?.bundleIdentifier ?? "nil")"
}

@MainActor
func diagCapture(
  _ name: String, section: String, isShown: () -> Bool, panelVisible: () -> Bool, panelFrame: () -> NSRect,
  popoverVisible: () -> Bool, popoverFrame: () -> NSRect, target: () -> String
) {
  func state() -> String {
    "popover.isShown=\\(isShown()) popoverWindowVisible=\\(popoverVisible()) panel.isVisible=\\(panelVisible()) panelFrame=\\(panelFrame()) popoverFrame=\\(popoverFrame()) target=\\(target())"
  }
  diagEmit("CAPTURE [\\(section)] \\(name) at \\(Date())")
  diagEmit("  now: \\(state())")
  diagEmit("  session: \\(diagSession())")
  let opening = String(name.split(separator: ":").first ?? "")
  guard !diagExtended.contains(opening) else { return }
  diagExtended.insert(opening)
  diagEmit("  trace before the capture:")
  for traced in diagTraceLines.suffix(40) { diagEmit("    " + traced) }
  let seen = diagTraceLines.count
  let start = Date()
  for wait in [0.5, 1, 2, 3, 5, 8] {
    drain(max(0, wait - Date().timeIntervalSince(start)))
    diagEmit("  later +\\(wait) s: \\(state())")
  }
  diagEmit("  trace during the longer drain:")
  for traced in diagTraceLines.dropFirst(seen) { diagEmit("    " + traced) }
  diagEmit("  session after: \\(diagSession())")
}
'''
